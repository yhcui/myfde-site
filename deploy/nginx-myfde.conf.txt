# myfde.top nginx 配置
# 用法：修改你服务器上现有的两个 server 块（主站 + job.myfde.top），保留各自的
#       listen / ssl_certificate 行，其余按下面替换。改完: nginx -t && systemctl reload nginx
# 前提：站点已在服务器上: git clone -b deploy https://gitee.com/<用户名>/myfde-site.git /var/www/myfde

# ============================================================
# 1) 主站 server 块（myfde.top）：改 root，加 location 段
# ============================================================
server {
    listen 80;
    listen 443 ssl;
    server_name myfde.top www.myfde.top;

    # ——保留你原有的 ssl_certificate / ssl_certificate_key 行——

    # 【改】root 指向新站点目录
    root /var/www/myfde;
    index index.html;

    # 【追加】声明字符集：llms.txt 等纯文本文件防止浏览器猜错编码变乱码
    charset utf-8;

    # 【追加】旧 URL 301（保住搜索引擎已有收录）
    # ⚠ 绝对不要加 `location = /index.html { return 301 /; }`
    #   它会被 index 模块的内部跳转命中，导致首页 301 死循环
    location = /fde.html    {
        if ($arg_slug) { return 301 /fde/$arg_slug/; }
        return 301 /fde/;
    }
    location = /agent.html  {
        if ($arg_slug) { return 301 /ai/$arg_slug/; }
        return 301 /ai/;
    }
    location = /tutorials.html { return 301 /fde/training/; }
    location = /about.html     { return 301 /about/; }
    location = /posts/         { return 301 /fde/; }

    # 【追加】安全：屏蔽 .git
    location ~ /\.git { deny all; return 404; }

    # 【追加】静态资源缓存
    location ~* \.(css|svg|woff2?|png|jpg|xml|txt)$ {
        expires 30d;
        add_header Cache-Control "public";
    }

    # 【追加】404 页
    error_page 404 /404.html;

    location / {
        try_files $uri $uri/ =404;
    }
}

# ============================================================
# 2) job.myfde.top server 块：整块替换成 301（保留证书行）
# ============================================================
server {
    listen 80;
    listen 443 ssl;
    server_name job.myfde.top;

    # ——保留你原有的 ssl_certificate / ssl_certificate_key 行——

    location = / {
        if ($arg_slug) { return 301 https://myfde.top/fde/jobs/$arg_slug/; }
        return 301 https://myfde.top/fde/jobs/;
    }
    location / {
        return 301 https://myfde.top/fde/jobs/;
    }
}

# ============================================================
# 3) 验证清单（reload 后逐条执行）
# ============================================================
# curl -I https://myfde.top/                          -> 200
# curl -I "https://myfde.top/fde.html?slug=fde-intro" -> 301 -> /fde/fde-intro/
# curl -I https://myfde.top/tutorials.html            -> 301 -> /fde/training/
# curl -I https://job.myfde.top/                      -> 301 -> https://myfde.top/fde/jobs/
# curl -I https://myfde.top/nonexistent               -> 404
